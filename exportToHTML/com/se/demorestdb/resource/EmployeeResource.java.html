1    package com.se.demorestdb.resource;
2    
3    import com.se.demorestdb.model.Employee;
4    import com.se.demorestdb.repo.EmployeeRepo;
5    import com.se.demorestdb.service.EmployeeService;
6    import jakarta.inject.Inject;
7    import jakarta.ws.rs.*;
8    import jakarta.ws.rs.core.MediaType;
9    import jakarta.ws.rs.core.Response;
10   
11   import java.util.List;
12   
13   @Path("/employee")
14   @Produces(MediaType.APPLICATION_JSON)
15   public class EmployeeResource {
16   
17       @Inject
18       private EmployeeService employeeService;
19   
20       @GET
21       @Produces(MediaType.APPLICATION_JSON)
22       public Response getAllEmployees() {
23           List<Employee> employees = employeeService.getAllEmployees();
24           return Response.ok(employees).build();
25       }
26   
27       @GET
28       @Path("/{id}")
29       @Produces(MediaType.APPLICATION_JSON)
30       public Response getEmployeeById(@PathParam("id") int id) {
31           Employee employee = employeeService.getEmployeeById(id);
32           if (employee == null) {
33               return Response.status(Response.Status.NOT_FOUND).build();
34           }
35           return Response.ok(employee).build();
36       }
37       @POST
38       @Consumes(MediaType.APPLICATION_JSON)
39       @Produces(MediaType.APPLICATION_JSON)
40       public Response addEmployee(Employee employee) {
41           Employee newEmployee = employeeService.addEmployee(employee);
42           return Response.status(Response.Status.CREATED).entity(newEmployee).build();
43       }
44       @PUT
45       @Path("/{id}")
46       @Consumes(MediaType.APPLICATION_JSON)
47       @Produces(MediaType.APPLICATION_JSON)
48       public Response updateEmployee(@PathParam("id") int id, Employee employee) {
49           Employee updatedEmployee = employeeService.updateEmployee(id, employee);
50           if (updatedEmployee == null) {
51               return Response.status(Response.Status.NOT_FOUND).build();
52           }
53           return Response.ok(updatedEmployee).build();
54       }
55   }
